    const PrivexMedia = {
      triggerUploadPrompt() {

      handleFileSelected(event) {
        const file = event.target.files[0];
        if (!file || !PrivexState.activeChatId) return;

        const reader = new FileReader();
        reader.onload = (e) => {
          const isImage = file.type.startsWith('image/');
          const newMsg = {
            id: 'm_' + Date.now(),
            senderId: PrivexState.currentUser.id,
            senderName: PrivexState.currentUser.displayName,
            text: isImage ? 'Shared an encrypted image' : `Attached file: ${file.name} (${(file.size / 1024).toFixed(1)} KB)`,
            mediaUrl: isImage ? e.target.result : null,
            timestamp: Date.now(),
            status: 'delivered'
          };

          if (!PrivexState.messages[PrivexState.activeChatId]) PrivexState.messages[PrivexState.activeChatId] = [];
          PrivexState.messages[PrivexState.activeChatId].push(newMsg);

          const chat = PrivexState.chats.find(c => c.id === PrivexState.activeChatId);
          if (chat) {
            chat.lastUpdated = Date.now();
            chat.lastSnippet = newMsg.text;
          }

          saveStateToDisk();
          PrivexChat.renderMessages(PrivexState.activeChatId);
          PrivexChat.scrollToBottom();
          PrivexToast.show('Media encrypted and shared', 'success');
        };

        if (file.type.startsWith('image/')) reader.readAsDataURL(file);
        else reader.readAsArrayBuffer(file);
      },

      recordVoiceSim() {
        if (!PrivexState.activeChatId) return;
        PrivexToast.show('Recording audio sample...', 'info');

        setTimeout(() => {
          const newMsg = {
            id: 'm_' + Date.now(),
            senderId: PrivexState.currentUser.id,
            senderName: PrivexState.currentUser.displayName,
            text: 'Voice message (0:08)',
            voiceNote: true,
            timestamp: Date.now(),
            status: 'delivered'
          };

          saveStateToDisk();
          PrivexChat.renderMessages(PrivexState.activeChatId);
          PrivexChat.scrollToBottom();
          PrivexToast.show('Voice message encrypted and sent', 'success');

      playVoiceSample(btn) {

    // 9. UI ORCHESTRATION & VIEW CONTROLLER
    const PrivexUI = {
      init() {
        initializeSeedData();
        this.renderChatList();
        this.renderSuggestedContacts();

        // Restore theme
        const savedTheme = PrivexStorage.get('theme', 'dark');
        this.applyTheme(savedTheme);

        // Auto select first chat on larger screens
        if (window.innerWidth > 720 && PrivexState.chats.length > 0) {
          PrivexChat.selectChat(PrivexState.chats[0].id);
        }

      renderChatList(filterText = '', filterCat = 'all') {
        const container = document.getElementById('chatListContainer');
        container.innerHTML = '';

        let list = [...PrivexState.chats];

        // Filter category
        if (filterCat === 'direct') list = list.filter(c => c.type === 'direct');
        else if (filterCat === 'groups') list = list.filter(c => c.type === 'private_group');
        else if (filterCat === 'public') list = list.filter(c => c.type === 'public');

        // Filter text
        if (filterText) {
          const q = filterText.toLowerCase();
          list = list.filter(c => c.title.toLowerCase().includes(q) || (c.handle && c.handle.toLowerCase().includes(q)));
        }

        // Sort by last updated descending
        list.sort((a, b) => (b.lastUpdated || 0) - (a.lastUpdated || 0));

        if (list.length === 0) {
          container.innerHTML = `
            <div style="padding: 24px; text-align: center; color: var(--text-muted); font-size: 13px;">
              No matching private conversations found.
            </div>
          `;
          return;
        }

        list.forEach(chat => {
          const row = document.createElement('div');
          row.className = `chat-row ${chat.id === PrivexState.activeChatId ? 'active' : ''}`;
          row.onclick = () => PrivexChat.selectChat(chat.id);

          const lastMsgs = PrivexState.messages[chat.id] || [];
          const lastMsg = lastMsgs[lastMsgs.length - 1];
          const snippet = lastMsg ? lastMsg.text : (chat.description || 'Encrypted room');

          row.innerHTML = `
            <div class="avatar-wrapper">
              <div class="avatar">${PrivexUI.escapeHtml(chat.avatarText)}</div>
              <div class="status-dot ${chat.online ? '' : 'offline'}"></div>
            </div>
            <div class="chat-row-meta">
              <div class="chat-row-header">
                <span class="chat-row-name">
                  ${PrivexUI.escapeHtml(chat.title)}
                  ${chat.verified ? '<span class="badge-verified">✓</span>' : ''}
                  ${chat.type === 'private_group' ? '<span class="badge-lock">🔒</span>' : ''}
                </span>
                <span class="chat-row-time">${chat.lastUpdated ? PrivexUI.formatTime(chat.lastUpdated) : ''}</span>
              </div>
              <div class="chat-row-preview">
                <span style="overflow:hidden; text-overflow:ellipsis; white-space:nowrap; flex:1;">${PrivexUI.escapeHtml(snippet)}</span>
                ${chat.unreadCount > 0 ? `<span class="unread-badge">${chat.unreadCount}</span>` : ''}
              </div>
            </div>
          `;
          container.appendChild(row);
        });
      },

      escapeHtml(str) {
        if (!str) return '';
        return str.replace(/[&<>'"]/g, tag => ({
          '&': '&amp;',
          '<': '&lt;',
          '>': '&gt;',
          "'": '&#39;',
          '"': '&quot;'
        }[tag] || tag));
      },

        const d = new Date(ts);
        return d.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
